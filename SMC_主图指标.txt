{--- 参数设置说明 (请在通达信公式编辑器上方参数栏填入以下参数) ---
参数名      默认值   最小值   最大值   步长
SWING_L     50       5        200      1      (大级别Swing波段周期)
INT_L       5        2        30       1      (次级别Internal结构周期)
SHOW_SWG    1        0        1        1      (是否显示Swing结构 BOS/CHoCH)
SHOW_INT    1        0        1        1      (是否显示Internal内部结构)
SHOW_OB     1        0        1        1      (是否显示Order Block订单块)
SHOW_FVG    1        0        1        1      (是否显示FVG公允价值缺口)
SHOW_EQ     1        0        1        1      (是否显示EQH/EQL流动性)
SHOW_SW     1        0        1        1      (是否显示强弱高低点Strong/Weak)
SHOW_PD     1        0        1        1      (是否显示折溢价均衡线EQ)
EQ_L        3        1        30       1      (EQH/EQL等高低点确认周期, 对应Pine的Bars Confirmation)
EQ_TS       0.1      0        0.5      0.1    (EQH/EQL阈值系数, 相对ATR200, 对应Pine的Threshold)
OB_MIT      0        0        1        1      (OB失效判定源: 0=用最高/最低价 1=用收盘价)
------------------------------------------------------------------------}

{ 波动率测算 ATR (与 Pine 的 ta.atr(200) 口径对齐) }
TR_VAL := MAX(MAX(HIGH - LOW, ABS(REF(CLOSE, 1) - HIGH)), ABS(REF(CLOSE, 1) - LOW));
ATR_200 := MA(TR_VAL, 200);

{========================================================================
 2. 波段分型与极值识别 (Swing & Internal Swings)
========================================================================}
{ Swing 大波段高低极值判定 }
SWG_NEW_H := REF(H, SWING_L) > HHV(H, SWING_L);
SWG_NEW_L := REF(L, SWING_L) < LLV(L, SWING_L);

SWG_LAST_NH := BARSLAST(SWG_NEW_H);
SWG_LAST_NL := BARSLAST(SWG_NEW_L);
SWG_LEG := IF(SWG_LAST_NH < SWG_LAST_NL, 0, 1);

SWG_PIV_H := SWG_LEG <> REF(SWG_LEG, 1) AND SWG_LEG = 0;
SWG_PIV_L := SWG_LEG <> REF(SWG_LEG, 1) AND SWG_LEG = 1;

SWG_H_BAR := BARSLAST(SWG_PIV_H) + SWING_L;
SWG_L_BAR := BARSLAST(SWG_PIV_L) + SWING_L;

SWG_HI := REF(H, SWG_H_BAR);
SWG_LO := REF(L, SWG_L_BAR);

{ Internal 次级别波段高低极值判定 }
INT_NEW_H := REF(H, INT_L) > HHV(H, INT_L);
INT_NEW_L := REF(L, INT_L) < LLV(L, INT_L);

INT_LAST_NH := BARSLAST(INT_NEW_H);
INT_LAST_NL := BARSLAST(INT_NEW_L);
INT_LEG := IF(INT_LAST_NH < INT_LAST_NL, 0, 1);

INT_PIV_H := INT_LEG <> REF(INT_LEG, 1) AND INT_LEG = 0;
INT_PIV_L := INT_LEG <> REF(INT_LEG, 1) AND INT_LEG = 1;

INT_H_BAR := BARSLAST(INT_PIV_H) + INT_L;
INT_L_BAR := BARSLAST(INT_PIV_L) + INT_L;

INT_HI := REF(H, INT_H_BAR);
INT_LO := REF(L, INT_L_BAR);

{========================================================================
 3. 结构突破与反转识别 (BOS & CHoCH)
========================================================================}
{ Swing 突破与趋势状态跟踪 }
SW_CROSS_UP := CROSS(C, SWG_HI);
SW_CROSS_DN := CROSS(SWG_LO, C);

SWG_HI_CHG := SWG_H_BAR <> REF(SWG_H_BAR, 1);
SWG_LO_CHG := SWG_L_BAR <> REF(SWG_L_BAR, 1);

SW_FIRST_UP := SW_CROSS_UP AND SUM(SW_CROSS_UP, BARSLAST(SWG_HI_CHG) + 1) = 1;
SW_FIRST_DN := SW_CROSS_DN AND SUM(SW_CROSS_DN, BARSLAST(SWG_LO_CHG) + 1) = 1;

SW_LAST_UP := BARSLAST(SW_FIRST_UP);
SW_LAST_DN := BARSLAST(SW_FIRST_DN);
SW_TREND := IF(SW_LAST_UP < SW_LAST_DN, 1, IF(SW_LAST_DN < SW_LAST_UP, -1, 0));

SW_BULL_CHOCH := SHOW_SWG AND SW_FIRST_UP AND REF(SW_TREND, 1) = -1;
SW_BULL_BOS   := SHOW_SWG AND SW_FIRST_UP AND (REF(SW_TREND, 1) = 1 OR REF(SW_TREND, 1) = 0);
SW_BEAR_CHOCH := SHOW_SWG AND SW_FIRST_DN AND REF(SW_TREND, 1) = 1;
SW_BEAR_BOS   := SHOW_SWG AND SW_FIRST_DN AND (REF(SW_TREND, 1) = -1 OR REF(SW_TREND, 1) = 0);

{ Internal 突破与趋势状态跟踪 }
INT_CROSS_UP := CROSS(C, INT_HI);
INT_CROSS_DN := CROSS(INT_LO, C);

INT_HI_CHG := INT_H_BAR <> REF(INT_H_BAR, 1);
INT_LO_CHG := INT_L_BAR <> REF(INT_L_BAR, 1);

INT_FIRST_UP := INT_CROSS_UP AND SUM(INT_CROSS_UP, BARSLAST(INT_HI_CHG) + 1) = 1;
INT_FIRST_DN := INT_CROSS_DN AND SUM(INT_CROSS_DN, BARSLAST(INT_LO_CHG) + 1) = 1;

INT_LAST_UP := BARSLAST(INT_FIRST_UP);
INT_LAST_DN := BARSLAST(INT_FIRST_DN);
INT_TREND := IF(INT_LAST_UP < INT_LAST_DN, 1, IF(INT_LAST_DN < INT_LAST_UP, -1, 0));

INT_BULL_CHOCH := SHOW_INT AND INT_FIRST_UP AND INT_HI <> SWG_HI AND REF(INT_TREND, 1) = -1;
INT_BULL_BOS   := SHOW_INT AND INT_FIRST_UP AND INT_HI <> SWG_HI AND (REF(INT_TREND, 1) = 1 OR REF(INT_TREND, 1) = 0);
INT_BEAR_CHOCH := SHOW_INT AND INT_FIRST_DN AND INT_LO <> SWG_LO AND REF(INT_TREND, 1) = 1;
INT_BEAR_BOS   := SHOW_INT AND INT_FIRST_DN AND INT_LO <> SWG_LO AND (REF(INT_TREND, 1) = -1 OR REF(INT_TREND, 1) = 0);

{ 绘制 Swing 突破与反转标注 (大字/实线) }
DRAWTEXT(SW_BULL_CHOCH, H * 1.014, '★Swing CHoCH'), COLOR00FFFF;
DRAWTEXT(SW_BULL_BOS,   H * 1.012, '▲Swing BOS'),   COLORRED;
DRAWTEXT(SW_BEAR_CHOCH, L * 0.986, '★Swing CHoCH'), COLORFFFF00;
DRAWTEXT(SW_BEAR_BOS,   L * 0.988, '▼Swing BOS'),   COLORGREEN;

{ 绘制 Internal 突破与反转标注 (小字) }
DRAWTEXT(INT_BULL_CHOCH AND NOT(SW_BULL_CHOCH) AND NOT(SW_BULL_BOS), H * 1.010, 'i-CHoCH'), COLORFF88FF;
DRAWTEXT(INT_BULL_BOS   AND NOT(SW_BULL_CHOCH) AND NOT(SW_BULL_BOS), H * 1.008, 'i-BOS'),   COLORFF88FF;
DRAWTEXT(INT_BEAR_CHOCH AND NOT(SW_BEAR_CHOCH) AND NOT(SW_BEAR_BOS), L * 0.990, 'i-CHoCH'), COLOR88FFFF;
DRAWTEXT(INT_BEAR_BOS   AND NOT(SW_BEAR_CHOCH) AND NOT(SW_BEAR_BOS), L * 0.992, 'i-BOS'),   COLOR88FFFF;

{ 结构连接线: 与TradingView一致, 从被突破的分型K线画到突破K线 (Swing实线/Internal虚线) }
DRAWLINE(SHOW_SWG AND BACKSET(SWG_PIV_H, SWING_L + 1) AND NOT(REF(BACKSET(SWG_PIV_H, SWING_L + 1), 1)), H, SHOW_SWG AND SW_FIRST_UP, SWG_HI, 0), COLORRED, LINETHICK1;
DRAWLINE(SHOW_SWG AND BACKSET(SWG_PIV_L, SWING_L + 1) AND NOT(REF(BACKSET(SWG_PIV_L, SWING_L + 1), 1)), L, SHOW_SWG AND SW_FIRST_DN, SWG_LO, 0), COLORGREEN, LINETHICK1;
DRAWLINE(SHOW_INT AND BACKSET(INT_PIV_H, INT_L + 1) AND NOT(REF(BACKSET(INT_PIV_H, INT_L + 1), 1)), H, SHOW_INT AND INT_FIRST_UP, INT_HI, 0), COLORFF5555, DOTLINE;
DRAWLINE(SHOW_INT AND BACKSET(INT_PIV_L, INT_L + 1) AND NOT(REF(BACKSET(INT_PIV_L, INT_L + 1), 1)), L, SHOW_INT AND INT_FIRST_DN, INT_LO, 0), COLOR55FF55, DOTLINE;

{========================================================================
 4. 订单块 (Order Blocks - OB)
========================================================================}
HI_VOL := (H - L) >= 2 * ATR_200;
P_H := IF(HI_VOL, L, H);
P_L := IF(HI_VOL, H, L);

{ 看涨 Swing OB: 自最近Swing高点(pivot)至当前, 取解析后最低K线 }
SW_BULL_OB_BAR := LLVBARS(P_L, SWG_H_BAR + 1);
SW_OB_B_TOP := REF(P_H, SW_BULL_OB_BAR);
SW_OB_B_BOT := REF(P_L, SW_BULL_OB_BAR);

{ 看跌 Swing OB: 自最近Swing低点(pivot)至当前, 取解析后最高K线 }
SW_BEAR_OB_BAR := HHVBARS(P_H, SWG_L_BAR + 1);
SW_OB_S_TOP := REF(P_H, SW_BEAR_OB_BAR);
SW_OB_S_BOT := REF(P_L, SW_BEAR_OB_BAR);

{ 看涨 Internal OB: 自最近Internal高点(pivot)至当前, 取解析后最低K线 }
INT_BULL_OB_BAR := LLVBARS(P_L, INT_H_BAR + 1);
INT_OB_B_TOP := REF(P_H, INT_BULL_OB_BAR);
INT_OB_B_BOT := REF(P_L, INT_BULL_OB_BAR);

{ 看跌 Internal OB: 自最近Internal低点(pivot)至当前, 取解析后最高K线 }
INT_BEAR_OB_BAR := HHVBARS(P_H, INT_L_BAR + 1);
INT_OB_S_TOP := REF(P_H, INT_BEAR_OB_BAR);
INT_OB_S_BOT := REF(P_L, INT_BEAR_OB_BAR);

{ 记录最近订单块周期 (Swing 与 Internal 分开, 对应 Pine 的两类 OB) }
LAST_SW_BULL_OB := BARSLAST(SW_FIRST_UP);
LAST_SW_BEAR_OB := BARSLAST(SW_FIRST_DN);
LAST_INT_BULL_OB := BARSLAST(INT_FIRST_UP);
LAST_INT_BEAR_OB := BARSLAST(INT_FIRST_DN);

SW_BULL_OB_ACTIVE := SHOW_OB AND IF(OB_MIT = 0, LLV(L, LAST_SW_BULL_OB + 1) >= REF(SW_OB_B_BOT, LAST_SW_BULL_OB), LLV(C, LAST_SW_BULL_OB + 1) >= REF(SW_OB_B_BOT, LAST_SW_BULL_OB));
SW_BEAR_OB_ACTIVE := SHOW_OB AND IF(OB_MIT = 0, HHV(H, LAST_SW_BEAR_OB + 1) <= REF(SW_OB_S_TOP, LAST_SW_BEAR_OB), HHV(C, LAST_SW_BEAR_OB + 1) <= REF(SW_OB_S_TOP, LAST_SW_BEAR_OB));
INT_BULL_OB_ACTIVE := SHOW_OB AND IF(OB_MIT = 0, LLV(L, LAST_INT_BULL_OB + 1) >= REF(INT_OB_B_BOT, LAST_INT_BULL_OB), LLV(C, LAST_INT_BULL_OB + 1) >= REF(INT_OB_B_BOT, LAST_INT_BULL_OB));
INT_BEAR_OB_ACTIVE := SHOW_OB AND IF(OB_MIT = 0, HHV(H, LAST_INT_BEAR_OB + 1) <= REF(INT_OB_S_TOP, LAST_INT_BEAR_OB), HHV(C, LAST_INT_BEAR_OB + 1) <= REF(INT_OB_S_TOP, LAST_INT_BEAR_OB));

{ 绘制订单块区间带 (颜色与价格结构标志一致: 多头红系/空头绿系) }
STICKLINE(SW_BULL_OB_ACTIVE, REF(SW_OB_B_BOT, LAST_SW_BULL_OB), REF(SW_OB_B_TOP, LAST_SW_BULL_OB), 2, 0), COLORRED;
STICKLINE(SW_BEAR_OB_ACTIVE, REF(SW_OB_S_BOT, LAST_SW_BEAR_OB), REF(SW_OB_S_TOP, LAST_SW_BEAR_OB), 2, 0), COLORGREEN;
STICKLINE(INT_BULL_OB_ACTIVE, REF(INT_OB_B_BOT, LAST_INT_BULL_OB), REF(INT_OB_B_TOP, LAST_INT_BULL_OB), 2, 0), COLORFF88FF;
STICKLINE(INT_BEAR_OB_ACTIVE, REF(INT_OB_S_BOT, LAST_INT_BEAR_OB), REF(INT_OB_S_TOP, LAST_INT_BEAR_OB), 2, 0), COLOR88FFFF;

{========================================================================
 5. 公允价值缺口 (Fair Value Gaps - FVG)
========================================================================}
FVG_BULL := SHOW_FVG AND L > REF(H, 2) AND REF(C, 1) > REF(H, 2);
FVG_BEAR := SHOW_FVG AND H < REF(L, 2) AND REF(C, 1) < REF(L, 2);

{ 绘制 FVG 缺口方块 }
STICKLINE(FVG_BULL, REF(H, 2), L, 3, 0), COLOR005522;
STICKLINE(FVG_BEAR, REF(L, 2), H, 3, 0), COLOR220055;

{ 标注 FVG 文本 }
DRAWTEXT(FVG_BULL, L * 0.995, 'FVG+'), COLOR00FF7F;
DRAWTEXT(FVG_BEAR, H * 1.005, 'FVG-'), COLOR7F7FFF;

{========================================================================
 6. 等高点与等低点 (Equal Highs & Lows - EQH / EQL 流动性池)
========================================================================}
EQ_NEW_H := REF(H, EQ_L) > HHV(H, EQ_L);
EQ_NEW_L := REF(L, EQ_L) < LLV(L, EQ_L);

EQ_LAST_NH := BARSLAST(EQ_NEW_H);
EQ_LAST_NL := BARSLAST(EQ_NEW_L);
EQ_LEG := IF(EQ_LAST_NH < EQ_LAST_NL, 0, 1);

EQ_PIV_H := EQ_LEG <> REF(EQ_LEG, 1) AND EQ_LEG = 0;
EQ_PIV_L := EQ_LEG <> REF(EQ_LEG, 1) AND EQ_LEG = 1;

EQ_H_BAR := BARSLAST(EQ_PIV_H) + EQ_L;
EQ_L_BAR := BARSLAST(EQ_PIV_L) + EQ_L;

EQ_HI := REF(H, EQ_H_BAR);
EQ_LO := REF(L, EQ_L_BAR);

PREV_EQ_HI := REF(EQ_HI, EQ_H_BAR + 1);
PREV_EQ_LO := REF(EQ_LO, EQ_L_BAR + 1);

IS_EQH := SHOW_EQ AND EQ_PIV_H AND ABS(REF(H, EQ_L) - PREV_EQ_HI) < EQ_TS * ATR_200;
IS_EQL := SHOW_EQ AND EQ_PIV_L AND ABS(REF(L, EQ_L) - PREV_EQ_LO) < EQ_TS * ATR_200;

DRAWTEXT(IS_EQH, H * 1.016, 'EQH (Liquidity)'), COLORFF33FF;
DRAWTEXT(IS_EQL, L * 0.984, 'EQL (Liquidity)'), COLOR33FFFF;

{========================================================================
 7. 强弱高低点 (Strong / Weak High & Low)
========================================================================}
{ 与 Pine 一致: 显示当前Swing腿的trailing极值水平线并延伸至最新bar,
  空头趋势时高点为Strong High/低点为Weak Low, 多头趋势时相反 }
CURR_RANGE_HI := HHV(H, SWG_H_BAR + 1);
CURR_RANGE_LO := LLV(L, SWG_L_BAR + 1);
CURR_EQ       := (CURR_RANGE_HI + CURR_RANGE_LO) / 2;

{ 当前腿极值水平线 (起点=最近极值bar, 终点=最新bar) }
DRAWLINE(SHOW_SW AND SW_TREND = -1 AND H = CURR_RANGE_HI, CURR_RANGE_HI, SHOW_SW AND ISLASTBAR AND SW_TREND = -1, CURR_RANGE_HI, 0), COLOR0099FF, LINETHICK1;
DRAWLINE(SHOW_SW AND SW_TREND = 1 AND H = CURR_RANGE_HI,  CURR_RANGE_HI, SHOW_SW AND ISLASTBAR AND SW_TREND = 1,  CURR_RANGE_HI, 0), COLOR888888, LINETHICK1;
DRAWLINE(SHOW_SW AND SW_TREND = 1 AND L = CURR_RANGE_LO,  CURR_RANGE_LO, SHOW_SW AND ISLASTBAR AND SW_TREND = 1,  CURR_RANGE_LO, 0), COLOR00FF99, LINETHICK1;
DRAWLINE(SHOW_SW AND SW_TREND = -1 AND L = CURR_RANGE_LO, CURR_RANGE_LO, SHOW_SW AND ISLASTBAR AND SW_TREND = -1, CURR_RANGE_LO, 0), COLOR888888, LINETHICK1;

{ 最新bar上标注文字 }
DRAWTEXT(SHOW_SW AND ISLASTBAR AND SW_TREND = -1, CURR_RANGE_HI * 1.005, 'Strong High'), COLOR0099FF;
DRAWTEXT(SHOW_SW AND ISLASTBAR AND SW_TREND = 1,  CURR_RANGE_HI * 1.005, 'Weak High'),   COLOR888888;
DRAWTEXT(SHOW_SW AND ISLASTBAR AND SW_TREND = 1,  CURR_RANGE_LO * 0.995, 'Strong Low'),  COLOR00FF99;
DRAWTEXT(SHOW_SW AND ISLASTBAR AND SW_TREND = -1, CURR_RANGE_LO * 0.995, 'Weak Low'),    COLOR888888;

{========================================================================
 8. 折溢价与均衡位 (Premium / Discount / Equilibrium)
========================================================================}
{ 在主图绘制均衡位 50% 水平线 }
EQ_LINE: IF(SHOW_PD, CURR_EQ, DRAWNULL), DOTLINE, COLORGRAY;

{ 屏幕左上角状态看板 }
DRAWTEXT_FIX(SHOW_SWG, 0.01, 0.02, 0, '● 大波段Swing趋势:'), COLORWHITE;
DRAWTEXT_FIX(SHOW_SWG AND SW_TREND = 1,  0.15, 0.02, 0, '多'), COLORRED;
DRAWTEXT_FIX(SHOW_SWG AND SW_TREND = -1, 0.15, 0.02, 1, '空'), COLORGREEN;
DRAWTEXT_FIX(SHOW_INT, 0.01, 0.06, 0, '● 次级Internal趋势:'), COLORWHITE;
DRAWTEXT_FIX(SHOW_INT AND INT_TREND = 1,  0.15, 0.06, 0, '多'), COLORFF88FF;
DRAWTEXT_FIX(SHOW_INT AND INT_TREND = -1, 0.15, 0.06, 1, '空'), COLOR88FFFF;
