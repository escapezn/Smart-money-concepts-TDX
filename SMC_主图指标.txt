{--- 参数设置说明 (请在通达信公式编辑器上方参数栏填入以下参数) ---
参数名      默认值   最小值   最大值   步长
SWING_L     50       5        200      1      (大级别Swing波段周期)
INT_L       5        2        30       1      (次级别Internal结构周期)
SHOW_SWG    1        0        1        1      (是否显示Swing结构 BOS/CHoCH)
SHOW_INT    1        0        1        1      (是否显示Internal内部结构)
SHOW_OB     1        0        1        1      (是否显示Order Block订单块)
SHOW_FVG    0        0        1        1      (是否显示FVG公允价值缺口)
FVG_AT      1        0        1        1      (FVG自动阈值过滤, 对应Pine的Auto Threshold)
FVG_TF      0        0        1440     1      (FVG计算周期分钟数, 0=当前周期, 对应Pine的Timeframe)
SHOW_EQ     1        0        1        1      (是否显示EQH/EQL流动性)
SHOW_SW     1        0        1        1      (是否显示强弱高低点Strong/Weak)
SHOW_PD     1        0        1        1      (是否显示折溢价均衡线EQ)
EQ_L        3        1        30       1      (EQH/EQL等高低点确认周期, 对应Pine的Bars Confirmation)
EQ_TS       0.1      0        0.5      0.1    (EQH/EQL阈值系数, 相对ATR200, 对应Pine的Threshold)
OB_MIT      0        0        1        1      (OB失效判定源: 0=用最高/最低价 1=用收盘价)
------------------------------------------------------------------------}

{ 波动率测算 ATR (与 Pine 的 ta.atr(200) RMA 平滑对齐) }
ATR_200 := SMA(TR, 200, 1);

{========================================================================
 2. 波段分型与极值识别 (Swing & Internal Swings)
========================================================================}
{ Swing 大波段高低极值判定 }
SWG_NEW_H := REF(H, SWING_L) > HHV(H, SWING_L);
SWG_NEW_L := REF(L, SWING_L) < LLV(L, SWING_L);
SWG_SET_H := SWG_NEW_H;
SWG_SET_L := SWG_NEW_L AND NOT(SWG_NEW_H);

SWG_LAST_NH := BARSLAST(SWG_SET_H);
SWG_LAST_NL := BARSLAST(SWG_SET_L);
SWG_NH_BARS := IF(SWG_LAST_NH >= 0, SWG_LAST_NH, 999999);
SWG_NL_BARS := IF(SWG_LAST_NL >= 0, SWG_LAST_NL, 999999);
SWG_LEG := IF(SWG_NL_BARS < SWG_NH_BARS, 1, 0);

SWG_PREV_LEG := IF(BARPOS > 1, REF(SWG_LEG, 1), 0);
SWG_PIV_H := SWG_LEG <> SWG_PREV_LEG AND SWG_LEG = 0;
SWG_PIV_L := SWG_LEG <> SWG_PREV_LEG AND SWG_LEG = 1;

SWG_H_BAR := BARSLAST(SWG_PIV_H) + SWING_L;
SWG_L_BAR := BARSLAST(SWG_PIV_L) + SWING_L;

SWG_HI := REF(H, SWG_H_BAR);
SWG_LO := REF(L, SWG_L_BAR);

{ Internal 次级别波段高低极值判定 }
INT_NEW_H := REF(H, INT_L) > HHV(H, INT_L);
INT_NEW_L := REF(L, INT_L) < LLV(L, INT_L);
INT_SET_H := INT_NEW_H;
INT_SET_L := INT_NEW_L AND NOT(INT_NEW_H);

INT_LAST_NH := BARSLAST(INT_SET_H);
INT_LAST_NL := BARSLAST(INT_SET_L);
INT_NH_BARS := IF(INT_LAST_NH >= 0, INT_LAST_NH, 999999);
INT_NL_BARS := IF(INT_LAST_NL >= 0, INT_LAST_NL, 999999);
INT_LEG := IF(INT_NL_BARS < INT_NH_BARS, 1, 0);

INT_PREV_LEG := IF(BARPOS > 1, REF(INT_LEG, 1), 0);
INT_PIV_H := INT_LEG <> INT_PREV_LEG AND INT_LEG = 0;
INT_PIV_L := INT_LEG <> INT_PREV_LEG AND INT_LEG = 1;

INT_H_BAR := BARSLAST(INT_PIV_H) + INT_L;
INT_L_BAR := BARSLAST(INT_PIV_L) + INT_L;

INT_HI := REF(H, INT_H_BAR);
INT_LO := REF(L, INT_L_BAR);

{========================================================================
 3. 结构突破与反转识别 (BOS & CHoCH)
========================================================================}
{ Swing 突破与趋势状态跟踪 (同一分型生命周期内只触发首次突破) }
SW_CROSS_UP := CROSS(C, SWG_HI);
SW_CROSS_DN := CROSS(SWG_LO, C);

SW_FIRST_UP := SW_CROSS_UP AND COUNT(SW_CROSS_UP, BARSLAST(SWG_PIV_H) + 1) = 1;
SW_FIRST_DN := SW_CROSS_DN AND COUNT(SW_CROSS_DN, BARSLAST(SWG_PIV_L) + 1) = 1;

SW_LAST_UP := BARSLAST(SW_FIRST_UP);
SW_LAST_DN := BARSLAST(SW_FIRST_DN);
SW_UP_BARS := IF(SW_LAST_UP >= 0, SW_LAST_UP, 999999);
SW_DN_BARS := IF(SW_LAST_DN >= 0, SW_LAST_DN, 999999);
SW_TREND := IF(SW_UP_BARS < SW_DN_BARS, 1, IF(SW_DN_BARS < SW_UP_BARS, -1, 0));

SW_PREV_TREND := IF(REF(SW_TREND, 1) = 1, 1, IF(REF(SW_TREND, 1) = -1, -1, 0));

SW_BULL_CHOCH := SHOW_SWG AND SWG_HI > 0 AND SW_FIRST_UP AND SW_PREV_TREND = -1;
SW_BULL_BOS   := SHOW_SWG AND SWG_HI > 0 AND SW_FIRST_UP AND (SW_PREV_TREND = 1 OR SW_PREV_TREND = 0);
SW_BEAR_CHOCH := SHOW_SWG AND SWG_LO > 0 AND SW_FIRST_DN AND SW_PREV_TREND = 1;
SW_BEAR_BOS   := SHOW_SWG AND SWG_LO > 0 AND SW_FIRST_DN AND (SW_PREV_TREND = -1 OR SW_PREV_TREND = 0);

{ 大级别突破严格有效判定 (空值防御: 大级别未就绪时稳定返回0) }
SW_HAS_BULL := IF(SWG_HI > 0, IF(SW_BULL_CHOCH OR SW_BULL_BOS, 1, 0), 0);
SW_HAS_BEAR := IF(SWG_LO > 0, IF(SW_BEAR_CHOCH OR SW_BEAR_BOS, 1, 0), 0);

{ Internal 突破与趋势状态跟踪 (同一分型生命周期内只触发首次突破) }
INT_CROSS_UP := CROSS(C, INT_HI);
INT_CROSS_DN := CROSS(INT_LO, C);

INT_FIRST_UP := INT_CROSS_UP AND COUNT(INT_CROSS_UP, BARSLAST(INT_PIV_H) + 1) = 1;
INT_FIRST_DN := INT_CROSS_DN AND COUNT(INT_CROSS_DN, BARSLAST(INT_PIV_L) + 1) = 1;

INT_LAST_UP := BARSLAST(INT_FIRST_UP);
INT_LAST_DN := BARSLAST(INT_FIRST_DN);
INT_UP_BARS := IF(INT_LAST_UP >= 0, INT_LAST_UP, 999999);
INT_DN_BARS := IF(INT_LAST_DN >= 0, INT_LAST_DN, 999999);
INT_TREND := IF(INT_UP_BARS < INT_DN_BARS, 1, IF(INT_DN_BARS < INT_UP_BARS, -1, 0));

INT_PREV_TREND := IF(REF(INT_TREND, 1) = 1, 1, IF(REF(INT_TREND, 1) = -1, -1, 0));

{ 当大波段 Swing 极值尚未生成时，次级别 Internal 独立有效；已生成时剔除重叠价位 }
INT_DIFF_HI := IF(SWG_HI > 0, INT_HI <> SWG_HI, 1);
INT_DIFF_LO := IF(SWG_LO > 0, INT_LO <> SWG_LO, 1);

INT_UP_SIG := INT_FIRST_UP AND INT_DIFF_HI;
INT_DN_SIG := INT_FIRST_DN AND INT_DIFF_LO;

INT_BULL_CHOCH := SHOW_INT AND INT_UP_SIG AND INT_PREV_TREND = -1;
INT_BULL_BOS   := SHOW_INT AND INT_UP_SIG AND (INT_PREV_TREND = 1 OR INT_PREV_TREND = 0);
INT_BEAR_CHOCH := SHOW_INT AND INT_DN_SIG AND INT_PREV_TREND = 1;
INT_BEAR_BOS   := SHOW_INT AND INT_DN_SIG AND (INT_PREV_TREND = -1 OR INT_PREV_TREND = 0);

{ 绘制 Swing 突破与反转标注 (实线) }
DRAWTEXT(SW_BULL_CHOCH, H * 1.014, '▲CHoCH'), COLORRED;
DRAWTEXT(SW_BULL_BOS,   H * 1.012, '▲BOS'),   COLORRED;
DRAWTEXT(SW_BEAR_CHOCH, L * 0.986, '▼CHoCH'), COLORGREEN;
DRAWTEXT(SW_BEAR_BOS,   L * 0.988, '▼BOS'),   COLORGREEN;

{ 绘制 Internal 突破与反转标注 (虚线) }
DRAWTEXT(INT_BULL_CHOCH AND SW_HAS_BULL = 0, H * 1.010, 'CHoCH'), COLORFF88FF;
DRAWTEXT(INT_BULL_BOS   AND SW_HAS_BULL = 0, H * 1.008, 'BOS'),   COLORFF88FF;
DRAWTEXT(INT_BEAR_CHOCH AND SW_HAS_BEAR = 0, L * 0.990, 'CHoCH'), COLOR88FFFF;
DRAWTEXT(INT_BEAR_BOS   AND SW_HAS_BEAR = 0, L * 0.992, 'BOS'),   COLOR88FFFF;

{ 结构连接线: 奇偶双通道独立绘制水平线 (Swing实线/Internal虚线, 与TradingView一致) }
SW_UP_CNT := SUM(SW_FIRST_UP, 0);
SW_UP_1 := SW_FIRST_UP AND MOD(SW_UP_CNT, 2) = 1;
SW_UP_2 := SW_FIRST_UP AND MOD(SW_UP_CNT, 2) = 0;
SW_START_UP1 := BACKSET(SW_UP_1, SWG_H_BAR + 1) AND NOT(REF(BACKSET(SW_UP_1, SWG_H_BAR + 1), 1));
SW_START_UP2 := BACKSET(SW_UP_2, SWG_H_BAR + 1) AND NOT(REF(BACKSET(SW_UP_2, SWG_H_BAR + 1), 1));

SW_DN_CNT := SUM(SW_FIRST_DN, 0);
SW_DN_1 := SW_FIRST_DN AND MOD(SW_DN_CNT, 2) = 1;
SW_DN_2 := SW_FIRST_DN AND MOD(SW_DN_CNT, 2) = 0;
SW_START_DN1 := BACKSET(SW_DN_1, SWG_L_BAR + 1) AND NOT(REF(BACKSET(SW_DN_1, SWG_L_BAR + 1), 1));
SW_START_DN2 := BACKSET(SW_DN_2, SWG_L_BAR + 1) AND NOT(REF(BACKSET(SW_DN_2, SWG_L_BAR + 1), 1));

DRAWLINE(SHOW_SWG AND SW_START_UP1, H, SHOW_SWG AND SW_UP_1, SWG_HI, 0), COLORRED, LINETHICK1;
DRAWLINE(SHOW_SWG AND SW_START_UP2, H, SHOW_SWG AND SW_UP_2, SWG_HI, 0), COLORRED, LINETHICK1;
DRAWLINE(SHOW_SWG AND SW_START_DN1, L, SHOW_SWG AND SW_DN_1, SWG_LO, 0), COLORGREEN, LINETHICK1;
DRAWLINE(SHOW_SWG AND SW_START_DN2, L, SHOW_SWG AND SW_DN_2, SWG_LO, 0), COLORGREEN, LINETHICK1;

INT_UP_CNT := SUM(INT_UP_SIG, 0);
INT_UP_1 := INT_UP_SIG AND MOD(INT_UP_CNT, 2) = 1;
INT_UP_2 := INT_UP_SIG AND MOD(INT_UP_CNT, 2) = 0;
INT_START_UP1 := BACKSET(INT_UP_1, INT_H_BAR + 1) AND NOT(REF(BACKSET(INT_UP_1, INT_H_BAR + 1), 1));
INT_START_UP2 := BACKSET(INT_UP_2, INT_H_BAR + 1) AND NOT(REF(BACKSET(INT_UP_2, INT_H_BAR + 1), 1));

INT_DN_CNT := SUM(INT_DN_SIG, 0);
INT_DN_1 := INT_DN_SIG AND MOD(INT_DN_CNT, 2) = 1;
INT_DN_2 := INT_DN_SIG AND MOD(INT_DN_CNT, 2) = 0;
INT_START_DN1 := BACKSET(INT_DN_1, INT_L_BAR + 1) AND NOT(REF(BACKSET(INT_DN_1, INT_L_BAR + 1), 1));
INT_START_DN2 := BACKSET(INT_DN_2, INT_L_BAR + 1) AND NOT(REF(BACKSET(INT_DN_2, INT_L_BAR + 1), 1));

DRAWLINE(SHOW_INT AND INT_START_UP1, H, SHOW_INT AND INT_UP_1, INT_HI, 0), COLORFF88FF, DOTLINE;
DRAWLINE(SHOW_INT AND INT_START_UP2, H, SHOW_INT AND INT_UP_2, INT_HI, 0), COLORFF88FF, DOTLINE;
DRAWLINE(SHOW_INT AND INT_START_DN1, L, SHOW_INT AND INT_DN_1, INT_LO, 0), COLOR88FFFF, DOTLINE;
DRAWLINE(SHOW_INT AND INT_START_DN2, L, SHOW_INT AND INT_DN_2, INT_LO, 0), COLOR88FFFF, DOTLINE;

{========================================================================
 4. 订单块 (Order Blocks - OB)
========================================================================}
{ 异常波动K线过滤 (对应 Pine 的 highVolatilityBar 解析) }
HI_VOL := (H - L) >= 2 * ATR_200;
P_H := IF(HI_VOL, L, H);
P_L := IF(HI_VOL, H, L);

{ 首次突破时刻精准锁定 OB 极值 K 线的区间与位置 (VALUEWHEN 机制) }
SW_OB_B_TOP_VAL := VALUEWHEN(SW_FIRST_UP, REF(P_H, LLVBARS(P_L, SWG_H_BAR + 1)));
SW_OB_B_BOT_VAL := VALUEWHEN(SW_FIRST_UP, REF(P_L, LLVBARS(P_L, SWG_H_BAR + 1)));
SW_OB_S_TOP_VAL := VALUEWHEN(SW_FIRST_DN, REF(P_H, HHVBARS(P_H, SWG_L_BAR + 1)));
SW_OB_S_BOT_VAL := VALUEWHEN(SW_FIRST_DN, REF(P_L, HHVBARS(P_H, SWG_L_BAR + 1)));

INT_OB_B_TOP_VAL := VALUEWHEN(INT_UP_SIG, REF(P_H, LLVBARS(P_L, INT_H_BAR + 1)));
INT_OB_B_BOT_VAL := VALUEWHEN(INT_UP_SIG, REF(P_L, LLVBARS(P_L, INT_H_BAR + 1)));
INT_OB_S_TOP_VAL := VALUEWHEN(INT_DN_SIG, REF(P_H, HHVBARS(P_H, INT_L_BAR + 1)));
INT_OB_S_BOT_VAL := VALUEWHEN(INT_DN_SIG, REF(P_L, HHVBARS(P_H, INT_L_BAR + 1)));

{ 记录最近订单块生成周期 }
LAST_SW_BULL_OB := BARSLAST(SW_FIRST_UP);
LAST_SW_BEAR_OB := BARSLAST(SW_FIRST_DN);
LAST_INT_BULL_OB := BARSLAST(INT_UP_SIG);
LAST_INT_BEAR_OB := BARSLAST(INT_DN_SIG);

{ OB 未失效状态跟踪 (Mitigation 检测: 价格回踩击穿区间下沿/上沿即失效) }
SW_B_MIT := IF(LAST_SW_BULL_OB >= 0, IF(OB_MIT = 0, LLV(L, LAST_SW_BULL_OB + 1) < SW_OB_B_BOT_VAL, LLV(C, LAST_SW_BULL_OB + 1) < SW_OB_B_BOT_VAL), 1);
SW_S_MIT := IF(LAST_SW_BEAR_OB >= 0, IF(OB_MIT = 0, HHV(H, LAST_SW_BEAR_OB + 1) > SW_OB_S_TOP_VAL, HHV(C, LAST_SW_BEAR_OB + 1) > SW_OB_S_TOP_VAL), 1);
INT_B_MIT := IF(LAST_INT_BULL_OB >= 0, IF(OB_MIT = 0, LLV(L, LAST_INT_BULL_OB + 1) < INT_OB_B_BOT_VAL, LLV(C, LAST_INT_BULL_OB + 1) < INT_OB_B_BOT_VAL), 1);
INT_S_MIT := IF(LAST_INT_BEAR_OB >= 0, IF(OB_MIT = 0, HHV(H, LAST_INT_BEAR_OB + 1) > INT_OB_S_TOP_VAL, HHV(C, LAST_INT_BEAR_OB + 1) > INT_OB_S_TOP_VAL), 1);

SW_BULL_OB_ACTIVE := SHOW_OB AND (LAST_SW_BULL_OB >= 0) AND NOT(SW_B_MIT);
SW_BEAR_OB_ACTIVE := SHOW_OB AND (LAST_SW_BEAR_OB >= 0) AND NOT(SW_S_MIT);
INT_BULL_OB_ACTIVE := SHOW_OB AND (LAST_INT_BULL_OB >= 0) AND NOT(INT_B_MIT);
INT_BEAR_OB_ACTIVE := SHOW_OB AND (LAST_INT_BEAR_OB >= 0) AND NOT(INT_S_MIT);

{ 绘制有效订单块区间带 (与 TradingView 颜色体系一致) }
STICKLINE(SW_BULL_OB_ACTIVE, SW_OB_B_BOT_VAL, SW_OB_B_TOP_VAL, 2, 0), COLORRED;
STICKLINE(SW_BEAR_OB_ACTIVE, SW_OB_S_BOT_VAL, SW_OB_S_TOP_VAL, 2, 0), COLORGREEN;
STICKLINE(INT_BULL_OB_ACTIVE, INT_OB_B_BOT_VAL, INT_OB_B_TOP_VAL, 2, 0), COLORFF88FF;
STICKLINE(INT_BEAR_OB_ACTIVE, INT_OB_S_BOT_VAL, INT_OB_S_TOP_VAL, 2, 0), COLOR88FFFF;

{========================================================================
 5. 公允价值缺口 (Fair Value Gaps - FVG)
========================================================================}
{ 目标周期分桶: FVG_TF=0 表示当前周期; 用日期+日内分钟数生成单调递增的周期编号 }
FVG_MINS := IF(TIME >= 10000, FLOOR(TIME / 10000) * 60 + FLOOR(MOD(TIME, 10000) / 100), FLOOR(TIME / 100) * 60 + MOD(TIME, 100));
FVG_BK := IF(FVG_TF > 0, DATE * (FLOOR(1440 / MAX(FVG_TF, 1)) + 1) + FLOOR(FVG_MINS / MAX(FVG_TF, 1)), 0);
FVG_T_NEW := IF(BARPOS = 1, 1, IF(FVG_TF > 0, FVG_BK <> REF(FVG_BK, 1), 1));

{ 当前周期桶的运行高低开 (对应Pine当前未完成HTF bar的lookahead数据) }
FVG_D := BARSLAST(FVG_T_NEW);
FVG_RUN_H := HHV(H, FVG_D + 1);
FVG_RUN_L := LLV(L, FVG_D + 1);
FVG_RUN_O := REF(O, FVG_D);

{ 前一个已完成周期桶 (对应Pine的 close[1]/open[1]/high[1]/low[1]) }
FVG_P1_H := REF(FVG_RUN_H, FVG_D + 1);
FVG_P1_L := REF(FVG_RUN_L, FVG_D + 1);
FVG_P1_O := REF(FVG_RUN_O, FVG_D + 1);
FVG_P1_C := REF(C, FVG_D + 1);

{ 前前个已完成周期桶 (对应Pine的 high[2]/low[2]) }
FVG_P2_H := REF(FVG_P1_H, FVG_D + 1);
FVG_P2_L := REF(FVG_P1_L, FVG_D + 1);

{ 中间K线/周期桶实体幅度 + Auto Threshold (对应Pine的 barDeltaPercent / cum / bar_index*2) }
FVG_DELTA := IF(BARPOS > 2 AND FVG_P1_O > 0, (FVG_P1_C - FVG_P1_O) / (FVG_P1_O * 100), 0);
FVG_CUM := SUM(IF(FVG_T_NEW, ABS(FVG_DELTA), 0), BARPOS);
FVG_THR := IF(BARPOS > 2, FVG_CUM / (BARPOS - 1) * 2, 0);
FVG_MIN := IF(FVG_AT = 1, FVG_THR, 0);

FVG_BULL := SHOW_FVG AND BARPOS > 5 AND FVG_T_NEW AND FVG_RUN_L > FVG_P2_H AND FVG_P1_C > FVG_P2_H AND FVG_DELTA > FVG_MIN;
FVG_BEAR := SHOW_FVG AND BARPOS > 5 AND FVG_T_NEW AND FVG_RUN_H < FVG_P2_L AND FVG_P1_C < FVG_P2_L AND -FVG_DELTA > FVG_MIN;

{ 绘制 FVG 缺口方块 }
STICKLINE(FVG_BULL, FVG_P2_H, FVG_RUN_L, 6, 0), COLORFDB8C0;
STICKLINE(FVG_BEAR, FVG_P2_L, FVG_RUN_H, 6, 0), COLORACF2BD;

{ 标注 FVG 文本 }
DRAWTEXT(FVG_BULL, FVG_RUN_L * 0.995, 'FVG+'), COLOR7F7FFF;
DRAWTEXT(FVG_BEAR, FVG_RUN_H * 1.005, 'FVG-'), COLOR00FF7F;

{========================================================================
 6. 等高点与等低点 (Equal Highs & Lows - EQH / EQL 流动性池)
========================================================================}
EQ_NEW_H := REF(H, EQ_L) > HHV(H, EQ_L);
EQ_NEW_L := REF(L, EQ_L) < LLV(L, EQ_L);
EQ_SET_H := EQ_NEW_H;
EQ_SET_L := EQ_NEW_L AND NOT(EQ_NEW_H);

EQ_LAST_NH := BARSLAST(EQ_SET_H);
EQ_LAST_NL := BARSLAST(EQ_SET_L);
EQ_NH_BARS := IF(EQ_LAST_NH >= 0, EQ_LAST_NH, 999999);
EQ_NL_BARS := IF(EQ_LAST_NL >= 0, EQ_LAST_NL, 999999);
EQ_LEG := IF(EQ_NL_BARS < EQ_NH_BARS, 1, 0);

EQ_PREV_LEG := IF(BARPOS > 1, REF(EQ_LEG, 1), 0);
EQ_PIV_H := EQ_LEG <> EQ_PREV_LEG AND EQ_LEG = 0;
EQ_PIV_L := EQ_LEG <> EQ_PREV_LEG AND EQ_LEG = 1;

{ 当前分型极值与上一个分型极值的精确取值 (对齐 Pine equalHigh / equalLow 结构体) }
CURR_EQ_HI := REF(H, EQ_L);
CURR_EQ_LO := REF(L, EQ_L);
PREV_EQ_HI := REF(VALUEWHEN(EQ_PIV_H, CURR_EQ_HI), 1);
PREV_EQ_LO := REF(VALUEWHEN(EQ_PIV_L, CURR_EQ_LO), 1);

{ 上一个分型确认点距离当前确认点的周期跨度 }
PREV_PIV_H_DIST := REF(BARSLAST(EQ_PIV_H), 1) + 1;
PREV_PIV_L_DIST := REF(BARSLAST(EQ_PIV_L), 1) + 1;

{ 等高点/等低点判定 (与 Pine 的 Threshold * ATR 对齐, 增加微量容差补偿通达信短视口样本预热与浮点误差) }
IS_EQH := SHOW_EQ AND EQ_PIV_H AND PREV_PIV_H_DIST > 0 AND ABS(CURR_EQ_HI - PREV_EQ_HI) <= (EQ_TS * ATR_200 * 1.05 + 0.001);
IS_EQL := SHOW_EQ AND EQ_PIV_L AND PREV_PIV_L_DIST > 0 AND ABS(CURR_EQ_LO - PREV_EQ_LO) <= (EQ_TS * ATR_200 * 1.05 + 0.001);

{ 前一分型点与当前分型点定位 (BACKSET 回填) }
EQH_P1 := BACKSET(IS_EQH, PREV_PIV_H_DIST + EQ_L + 1) AND NOT(REF(BACKSET(IS_EQH, PREV_PIV_H_DIST + EQ_L + 1), 1));
EQH_P2 := BACKSET(IS_EQH, EQ_L + 1) AND NOT(REF(BACKSET(IS_EQH, EQ_L + 1), 1));

EQL_P1 := BACKSET(IS_EQL, PREV_PIV_L_DIST + EQ_L + 1) AND NOT(REF(BACKSET(IS_EQL, PREV_PIV_L_DIST + EQ_L + 1), 1));
EQL_P2 := BACKSET(IS_EQL, EQ_L + 1) AND NOT(REF(BACKSET(IS_EQL, EQ_L + 1), 1));

{ 绘制两个等高/等低分型之间的水平虚线连接线 (与 TradingView 一致) }
DRAWLINE(EQH_P1, H, EQH_P2, H, 0), COLORRED, DOTLINE;
DRAWLINE(EQL_P1, L, EQL_P2, L, 0), COLORGREEN, DOTLINE;

{ 在当前分型 K 线标注 EQH / EQL 文字 }
